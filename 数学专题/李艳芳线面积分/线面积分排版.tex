\subsubsection*{一、曲线积分}

\paragraph{1. 第一类曲线积分的定义}

设 $L$ 为 $xOy$ 面内的一条光滑曲线弧，函数 $f(x,y)$ 在 $L$ 上有界。在 $L$ 上任意插入点列 $M_1,M_2,\dots,M_{n-1}$ 把 $L$ 分成 $n$ 个小段，第 $i$ 个小段长度为 $\Delta s_i$，$(\xi_i,\eta_i)$ 为第 $i$ 个小段上任意取定的点。若当各小弧段长度的最大值 $\lambda\to0$ 时，和式 $\sum_{i=1}^n f(\xi_i,\eta_i)\Delta s_i$ 的极限总存在且与分法及取法无关，则称此极限为 $f(x,y)$ 在 $L$ 上的第一类曲线积分：

\[
\int_L f(x,y)\,ds = \lim_{\lambda\to0}\sum_{i=1}^n f(\xi_i,\eta_i)\Delta s_i.
\]

\paragraph{2. 第一类曲线积分的性质}

\begin{itemize}[itemsep=2pt]
\item 线性性：$\int_L[\alpha f+\beta g]ds = \alpha\int_L f\,ds + \beta\int_L g\,ds$
\item 可加性：若 $L=L_1\cup L_2$，则 $\int_L f\,ds = \int_{L_1}f\,ds + \int_{L_2}f\,ds$
\item 若 $f\equiv1$，则 $\int_L ds = l$（$l$ 为 $L$ 的长度）
\item 保序性：若 $f\le g$，则 $\int_L f\,ds \le \int_L g\,ds$；特别地 $|\int_L f\,ds| \le \int_L |f|\,ds$
\end{itemize}

\paragraph{3. 第一类曲线积分的计算}

基本方法：将积分弧段参数化后化为定积分。

\begin{itemize}[itemsep=4pt]
\item 参数方程 $x=\varphi(t),\,y=\psi(t)\;(\alpha\le t\le\beta)$：
\[
\int_L f(x,y)\,ds = \int_\alpha^\beta f(\varphi(t),\psi(t))\sqrt{[\varphi'(t)]^2+[\psi'(t)]^2}\,dt
\]
\item $y=\psi(x)\;(x_0\le x\le x_1)$：$\displaystyle\int_L f(x,y)\,ds = \int_{x_0}^{x_1} f(x,\psi(x))\sqrt{1+[\psi'(x)]^2}\,dx$
\item $x=\varphi(y)\;(y_0\le y\le y_1)$：类似
\item 极坐标 $r=r(\theta)\;(\alpha\le\theta\le\beta)$：
\[
\int_L f(x,y)\,ds = \int_\alpha^\beta f(r\cos\theta,r\sin\theta)\sqrt{[r(\theta)]^2+[r'(\theta)]^2}\,d\theta
\]
\end{itemize}

\subsubsection*{二、第二类曲线积分}

\paragraph{1. 定义}
设 $L$ 为 $xOy$ 面内从 $A$ 到 $B$ 的有向光滑曲线弧，函数 $P(x,y),Q(x,y)$ 在 $L$ 上有界。在 $L$ 上沿方向插入点列将 $L$ 分成 $n$ 个有向小弧段 $\widetilde{M_{i-1}M_i}$，$\Delta x_i=x_i-x_{i-1}$，$\Delta y_i=y_i-y_{i-1}$。若和式 $\sum_{i=1}^n P(\xi_i,\eta_i)\Delta x_i$ 的极限总存在且与分法及取法无关，则称此极限为 $P(x,y)$ 在 $L$ 上对坐标 $x$ 的曲线积分，记作 $\int_L P(x,y)\,dx$。类似定义 $\int_L Q(x,y)\,dy$。

\paragraph{2. 物理意义}
$\int_L P\,dx+Q\,dy = \int_L \mathbf{F}\cdot d\mathbf{r}$，表示变力 $\mathbf{F}=(P,Q)$ 沿 $L$ 所做的功。

\paragraph{3. 计算}
设 $L$ 参数方程 $x=\varphi(t),y=\psi(t)$，$t$ 从 $\alpha$ 到 $\beta$（$\alpha$ 对应起点，$\beta$ 对应终点）：
\[
\int_L P\,dx+Q\,dy = \int_\alpha^\beta\bigl[P(\varphi(t),\psi(t))\varphi'(t)+Q(\varphi(t),\psi(t))\psi'(t)\bigr]dt.
\]

\begin{prob}
计算 $\displaystyle\int_L xy\,dx$，其中 $L$ 为抛物线 $y^2=x$ 上从 $A(1,-1)$ 到 $B(1,1)$ 的一段弧。
\end{prob}

\paragraph{4. 两类曲线积分的联系}
\[
\int_L P\,dx+Q\,dy = \int_L(P\cos\alpha+Q\cos\beta)\,ds,\qquad
\int_\Gamma P\,dx+Q\,dy+R\,dz = \int_\Gamma(P\cos\alpha+Q\cos\beta+R\cos\gamma)\,ds,
\]
其中 $\cos\alpha,\cos\beta,\cos\gamma$ 为曲线在点处的切向量方向余弦。

\begin{prob}
把 $\int_L P(x,y)\,dx+Q(x,y)\,dy$ 化成对弧长的曲线积分，其中 $L$ 为沿上半圆周 $x^2+y^2=2x$ 从 $(0,0)$ 到 $(1,1)$ 的曲线弧。
\end{prob}

\paragraph{5. 曲线积分的应用}
第一类曲线积分：弧长、质量、重心、转动惯量。第二类曲线积分：变力做功、环流量。

\begin{prob}
质点 $P$ 沿以 $AB$ 为直径的半圆周从 $A(1,2)$ 运动到 $B(3,4)$，受变力 $\mathbf{F}$ 作用，$\mathbf{F}$ 的大小等于 $P$ 与原点距离，方向垂直于 $OP$ 且与 $y$ 轴正向夹角小于 $\pi/2$。求 $\mathbf{F}$ 对 $P$ 所做的功。

\subsubsection*{四、曲面积分}

\paragraph{1. 第一类曲面积分的定义与计算}
