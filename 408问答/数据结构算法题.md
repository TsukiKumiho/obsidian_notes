> 两个有序数组 `arr1`，`arr2`，长度分别为`n1`，`n2`，将其合并为一个新的有序数组`arr3`

```c++
void merge(int arr1[], int n1, int arr2[], int n2, int arr3[]){
	int i = 0, j = 0, k = 0;
	while(i < n1 && j < n2){
        if(arr1[i] < arr2[j])
            arr3[k++] = arr1[i++];
        else
            arr3[k++] = arr2[j++];
    }
    while(i < n1)
        arr3[k++] = arr1[i++];
    while(j < n2)
        arr3[k++] = arr2[j++];
}
```

> 合并两个有序链表

```c++
struct node {
    int data;
    node* next;
    node(int val = 0, node* nxt = nullptr) : data(val), next(nxt) {}
};

node* mergeTwoLists(node* l1, node* l2) {
    node* preHead = new node(-1);
    node* prev = preHead;
    while (l1 != nullptr && l2 != nullptr) {
        if (l1->data < l2->data) {
            prev->next = l1;
            l1 = l1->next;
        } else {
            prev->next = l2;
            l2 = l2->next;
        }
        prev = prev->next;
    }

    if (l1 == nullptr) prev->next = l2;
    else prev->next = l1;
    
    return preHead->next;
}
```

```c++
void delete_x(int nums[], int &size, int x){
    int slow = 0;
    for(int fast = 0; fast < size; ++ fast)
        if(nums[fast] != x)
            nums[slow++] = nums[fast];
    size = slow;
}
```

```c++
void quick_sort(int q[], int l, int r){
    if (l >= r) return;
    int i = l - 1, j = r + 1, x = q[l + r >> 1];
    while (i < j){
        do i ++ ; while (q[i] < x);		//while(q[++i] < x);
        do j -- ; while (q[j] > x);		//while(q[--j] > x);
        if (i < j) swap(q[i], q[j]);
    }
    quick_sort(q, l, j), quick_sort(q, j + 1, r);
}
```

> 给定一个长度为 `n` 的整数数列，以及一个整数 `k`，求出数列从小到大排序后的第 `k` 个数。

```c++
int quick_select(int q[], int l, int r, int k) {
    if (l >= r) return q[l];
    int i = l - 1, j = r + 1, x = q[l + r >> 1];
    while (i < j) {
        do i ++ ; while (q[i] < x);
        do j -- ; while (q[j] > x);
        if (i < j) swap(q[i], q[j]);
    }
    if (k <= j) return quick_select(q, l, j, k);
    else return quick_select(q, j + 1, r, k);
}
```

堆排序

```c++
void down(int nums[], int n, int i){
    while(true){
        int largest = i, left = 2 * i + 1, right = 2 * i + 2;
        
        if(left < n && nums[left] > nums[largest])	
            largest = left;
        if(right < n && nums[right] > nums[largest])	
            largest = right;
        
        if(largest == i)	break;
        swap(nums[i], nums[largest]);
        i = largest;
    }
}

void heapSort(int nums[], int n){
    for(int i = n / 2 - 1; i >= 0; i --)
        down(nums, n, i);
    
    for(int i = n - 1; i > 0; i --){
        swap(nums[0], nums[i]);
        down(nums, i, 0);
    }
}
```

```c++
const int N = 100010;
int tmp[N];				//tmp[r - l + 1];

void merge_sort(int q[], int l, int r){
    if (l >= r) return;

    int mid = l + r >> 1;
    merge_sort(q, l, mid);
    merge_sort(q, mid + 1, r);

    int k = 0, i = l, j = mid + 1;
    while (i <= mid && j <= r)
        if (q[i] <= q[j]) tmp[k ++ ] = q[i ++ ];
        else tmp[k ++ ] = q[j ++ ];

    while (i <= mid) tmp[k ++ ] = q[i ++ ];
    while (j <= r) tmp[k ++ ] = q[j ++ ];

    for (i = l, j = 0; i <= r; i ++, j ++ ) q[i] = tmp[j];
}
```

```c++
void reserve(int r[], int start, int end){
    while(start < end){
        int tmp = r[start];
        r[start] = r[end];
        r[end] = tmp;		//swap();
        start ++;
        end --;
    }
}
```
